# B2: Shop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the shop's backend to the Payload app: guest checkout for books and merch, payment through Paystack, delivery through Terminal Africa, a Shop staff role, a Stock page and a Book courier panel, so an order can go from a cart to Delivered before any public page exists.

**Architecture:** Everything lives inside the existing Payload 3 app. A new `orders` collection is written only by server code in `src/shop/`, reached through Payload endpoints (`/api/shop/...`, `/api/orders/:id/...`, `/api/paystack/webhook`, `/api/terminal/webhook`). Paystack and Terminal each sit behind a small client module with one interface, which the tests replace with fakes. Jobs on B1's existing queue send the emails, expire unpaid orders and remove customer details from dead ones. Two custom admin pieces (a Stock page and a courier panel on the order screen) are small React components registered in the Payload config.

**Tech Stack:** Payload 3.90.2 (`@payloadcms/db-postgres`, `@payloadcms/next`, `@payloadcms/ui`), Next.js 16.4.1, React 19.2.9, Postgres 17 in Docker, Vitest 4.0.18, TypeScript 5.9.3, Node 24. No new packages.

**Spec:** `docs/superpowers/specs/2026-10-27-b2-shop-design.md`. When this plan and the spec disagree, the spec wins; stop and ask.

```ts
import type { Access, FieldAccess } from 'payload'

export type StaffRole = 'admin' | 'editor' | 'shop'

type MaybeStaff = { role?: unknown; collection?: unknown } | null | undefined

/** The role of an account from the `users` collection, or undefined for anyone else. */
function roleOf(user: unknown): unknown {
  const candidate = user as MaybeStaff
  if (!candidate) return undefined
  if (candidate.collection !== undefined && candidate.collection !== 'users') return undefined
  return candidate.role
}

/** True for an Admin or Editor account: the people who work on content. Shop staff are not included. */
export function isStaffUser(user: unknown): boolean {
  const role = roleOf(user)
  return role === 'admin' || role === 'editor'
}

/** True for an Admin or Shop staff account: the people who work on orders and stock. */
export function isShopUser(user: unknown): boolean {
  const role = roleOf(user)
  return role === 'admin' || role === 'shop'
}

/** True for any staff account, whatever its role. Used for opening /admin, one's own account and edit locks. */
export function isAnyStaffUser(user: unknown): boolean {
  return isStaffUser(user) || isShopUser(user)
}

export function isAdminUser(user: unknown): boolean {
  return roleOf(user) === 'admin'
}
```

```ts
import type { CollectionConfig, GlobalConfig } from 'payload'

import { isStaffUser } from '.'

type HiddenRule = NonNullable<NonNullable<CollectionConfig['admin']>['hidden']>

/** The lists and settings shop staff work with. Each already has its own menu rule. */
const SHOP_SIDE = new Set(['orders', 'users', 'shop-settings'])

/** Hidden from anyone who is not an admin or editor; for those, the list's own rule still applies. */
export function hiddenUnlessContentStaff(existing: HiddenRule | undefined): HiddenRule {
  return (args) => {
    if (!isStaffUser(args.user)) return true
    return typeof existing === 'function' ? existing(args) : Boolean(existing)
  }
}

/**
 * Payload shows a list in the admin menu and dashboard to anyone allowed to read it, and visitors
 * may read published content, so a Shop staff account would otherwise see every content list
 * (read-only). Everything except the shop side is hidden from it, applied once over the config in
 * the same way `guardDeletes` is.
 */
export function hideContentFromShopStaff<T extends CollectionConfig | GlobalConfig>(entities: T[]): T[] {
  return entities.map((entity) => {
    if (SHOP_SIDE.has(entity.slug)) return entity
    const admin = { ...entity.admin, hidden: hiddenUnlessContentStaff(entity.admin?.hidden as HiddenRule | undefined) }
    return { ...entity, admin } as T
  })
}
```

```ts
import { toKobo } from '@/shop/money'
import { stateLabel } from '@/shop/states'

export type FlatArea = { areaName: string; states: string[]; priceNGN: number }

/** The flat delivery price for a state: its area's price, or the everywhere-else price, or none. */
export function flatPriceFor(areas: FlatArea[], elsewherePriceNGN: number | null, state: string): { priceKobo: number; area: string } | null {
  const area = areas.find((candidate) => candidate.states.includes(state))
  if (area) return { priceKobo: toKobo(area.priceNGN), area: area.areaName }
  if (typeof elsewherePriceNGN === 'number') return { priceKobo: toKobo(elsewherePriceNGN), area: 'Everywhere else' }
  return null
}

type AreaRow = { areaName?: string | null; states?: string[] | null }

/** A message naming the first state listed in two areas, or null when each state is in one area at most. */
export function overlappingStatesProblem(rows: AreaRow[] | null | undefined): string | null {
  const seen = new Map<string, string>()
  for (const row of rows ?? []) {
    const name = row.areaName ?? ''
    for (const state of row.states ?? []) {
      const earlier = seen.get(state)
      if (earlier !== undefined) return `${stateLabel(state)} is in both "${earlier}" and "${name}". A state can only be in one area.`
      seen.set(state, name)
    }
  }
  return null
}
```

```ts
export type HistoryRow = {
  id?: string | null
  at: string
  from?: string | null
  to?: string | null
  by: string
  note?: string | null
}

type NewEntry = { from?: string | null; to?: string | null; by: string; note?: string | null; at?: string }

/** The history with one entry added at the end. Existing rows keep their ids, so Payload keeps them. */
export function appendHistory(existing: HistoryRow[] | null | undefined, entry: NewEntry): HistoryRow[] {
  return [
    ...(existing ?? []),
    { at: entry.at ?? new Date().toISOString(), from: entry.from ?? null, to: entry.to ?? null, by: entry.by, note: entry.note ?? null },
  ]
}

const sameTime = (a: string, b: string) => new Date(a).getTime() === new Date(b).getTime()

function sameEntry(a: HistoryRow, b: HistoryRow | undefined): boolean {
  return (
    b !== undefined &&
    sameTime(a.at, b.at) &&
    (a.from ?? null) === (b.from ?? null) &&
    (a.to ?? null) === (b.to ?? null) &&
    a.by === b.by &&
    (a.note ?? null) === (b.note ?? null)
  )
}

/** True when `after` starts with every entry of `before`, unchanged: the history only grew. */
export function keepsHistory(before: HistoryRow[] | null | undefined, after: HistoryRow[] | null | undefined): boolean {
  const old = before ?? []
  const next = after ?? []
  return next.length >= old.length && old.every((entry, index) => sameEntry(entry, next[index]))
}
```

```ts
/**
 * The rules for changes staff make. The history only ever grows, and only server code adds to it:
 * a staff save must send it back unchanged (the admin does). The phone and address can only be
 * corrected while the order is Paid and no courier is booked. A status set by hand is written to

 */
const staffChangeRules: CollectionBeforeChangeHook = ({ context, data, operation, originalDoc, req }) => {
  if (operation !== 'update' || !originalDoc) return data
  const before = originalDoc.history as HistoryRow[] | undefined
  if ('history' in data && !keepsHistory(before, data.history)) {
    rejectField(req, 'orders', 'history', 'The history can only be added to, never changed.')
  }

  if ('history' in data && (data.history ?? []).length !== (before ?? []).length) {
    rejectField(req, 'orders', 'history', 'The history can only be added to by the shop itself.')
  }

  const phoneChanged = 'customerPhone' in data && (data.customerPhone ?? null) !== (originalDoc.customerPhone ?? null)
  const addressChanged = 'address' in data && !sameAddress(data.address, originalDoc.address)
  const waitingToBook = originalDoc.status === 'paid' && !originalDoc.shipment?.terminalShipmentId
  if ((phoneChanged || addressChanged) && !waitingToBook) {

      req,
      'orders',
      phoneChanged ? 'customerPhone' : 'address',
```

```ts
import type { PayloadRequest } from 'payload'

import type { Book, Merch } from '@/payload-types'
import { Refusal } from '@/refusal'
import { toKobo } from '@/shop/money'

export const MAX_LINES = 20
export const MAX_QUANTITY = 10

export type CartLine = { collection: 'books' | 'merch'; id: number; option: string | null; quantity: number }

export type PricedLine = {
  collection: 'books' | 'merch'
  id: number
  optionId: string | null
  optionName: string | null
  title: string
  unitPriceKobo: number
  weightGrams: number
  quantity: number
}

export type Totals = { itemsKobo: number; deliveryKobo: number; totalKobo: number }

const UNREADABLE = 'The cart could not be read. Refresh the page and try again.'

/** The cart as the browser sent it, checked for shape only. Prices and stock come from resolveCart. */
export function parseCart(value: unknown): CartLine[] {
  if (!Array.isArray(value) || value.length === 0) throw new Refusal('The cart is empty.')
  if (value.length > MAX_LINES) throw new Refusal(`A cart can hold up to ${MAX_LINES} different items.`)
  const seen = new Set<string>()
  return value.map((raw) => {
    const entry = (raw && typeof raw === 'object' ? raw : {}) as Record<string, unknown>
    const collection = entry.collection
    const id = Number(entry.id)
    const option = entry.option === undefined || entry.option === null || entry.option === '' ? null : entry.option
    if ((collection !== 'books' && collection !== 'merch') || !Number.isInteger(id) || id < 1) throw new Refusal(UNREADABLE)
    if (option !== null && (typeof option !== 'string' || option.length > 64)) throw new Refusal(UNREADABLE)
    const quantity = Number(entry.quantity)
    if (!Number.isInteger(quantity) || quantity < 1 || quantity > MAX_QUANTITY) {
      throw new Refusal(`You can buy up to ${MAX_QUANTITY} of each item.`)
    }
    const key = `${collection}:${id}:${option ?? ''}`
    if (seen.has(key)) throw new Refusal(UNREADABLE)
    seen.add(key)
    return { collection, id, option, quantity }
  })
}

const GONE = 'An item in the cart is no longer available. Remove it and try again.'

/** The published version of a book: a newer draft an editor is working on never changes the price. */
async function publishedBook(req: PayloadRequest, id: number): Promise<Book> {
  const book = await req.payload.findByID({ collection: 'books', id, depth: 0, draft: false, overrideAccess: true, disableErrors: true, req })
  if (!book || book._status !== 'published') throw new Refusal(GONE)
  return book
}

async function publishedMerch(req: PayloadRequest, id: number): Promise<Merch> {
  const item = await req.payload.findByID({ collection: 'merch', id, depth: 0, draft: false, overrideAccess: true, disableErrors: true, req })
  if (!item || item._status !== 'published') throw new Refusal(GONE)
  return item
}

/** Every line looked up in the database: refused by name if it is not for sale or sold out, priced if it is. */
export async function resolveCart(req: PayloadRequest, lines: CartLine[]): Promise<PricedLine[]> {
  const priced: PricedLine[] = []
  for (const line of lines) {
    if (line.collection === 'books') {
      const book = await publishedBook(req, line.id)
      if (!book.sellInShop) throw new Refusal(`"${book.title}" is not for sale.`)
      const format = (book.formats ?? []).find((candidate) => candidate.id === line.option)
      if (!format) throw new Refusal(`Pick a format for "${book.title}".`)
      if (!format.inStock) throw new Refusal(`"${book.title}" (${format.formatName}) is sold out.`)
      priced.push({
        collection: 'books',
        id: book.id,
        optionId: format.id ?? null,
        optionName: format.formatName,
        title: book.title,
        unitPriceKobo: toKobo(format.priceNGN),
        weightGrams: format.weightGrams,
        quantity: line.quantity,
      })
      continue
    }
    const item = await publishedMerch(req, line.id)
    const options = item.options ?? []
    let optionId: string | null = null
    let optionName: string | null = null
    if (options.length > 0) {
      const option = options.find((candidate) => candidate.id === line.option)
      if (!option) throw new Refusal(`Pick an option for "${item.title}".`)
      if (!option.inStock) throw new Refusal(`"${item.title}" (${option.name}) is sold out.`)
      optionId = option.id ?? null
      optionName = option.name
    } else if (!item.inStock) {
      throw new Refusal(`"${item.title}" is sold out.`)
    }
    priced.push({
      collection: 'merch',
      id: item.id,
      optionId,
      optionName,
      title: item.title,
      unitPriceKobo: toKobo(item.priceNGN),
      weightGrams: item.weightGrams,
      quantity: line.quantity,
    })
  }
  return priced
}

/** The order's sums in kobo. `booksDiscountPercent` is B3's member discount on books; nothing passes it yet. */
export function priceOrder(lines: PricedLine[], deliveryKobo: number, { booksDiscountPercent = 0 }: { booksDiscountPercent?: number } = {}): Totals {
  const itemsKobo = lines.reduce((sum, line) => {
    const unit = line.collection === 'books' ? Math.round((line.unitPriceKobo * (100 - booksDiscountPercent)) / 100) : line.unitPriceKobo
    return sum + unit * line.quantity
  }, 0)
  return { itemsKobo, deliveryKobo, totalKobo: itemsKobo + deliveryKobo }
}

/** Every copy's weight plus the box. */
export function parcelWeightGrams(lines: { weightGrams: number; quantity: number }[], boxWeightGrams: number): number {
  return lines.reduce((sum, line) => sum + line.weightGrams * line.quantity, boxWeightGrams)
}
```

```ts
import { createHmac, timingSafeEqual } from 'node:crypto'

export function hmacSha512Hex(body: string, secret: string): string {
  return createHmac('sha512', secret).update(body).digest('hex')
}

/**
 * Paystack (and, assumed until the sandbox shows otherwise, Terminal) sign each webhook with the
 * HMAC-SHA512 of the raw body, made with the account's secret key, in hex. Compared in constant time.
 */
export function signatureMatches(body: string, sent: string | null, secret: string | undefined): boolean {
  if (!sent || !secret) return false
  const expected = Buffer.from(hmacSha512Hex(body, secret))
  const given = Buffer.from(sent.trim().toLowerCase())
  return expected.length === given.length && timingSafeEqual(expected, given)
}
```

```ts
const BASE_URL = 'https://api.paystack.co'
const TIMEOUT_MS = 10_000

export class PaystackError extends Error {
  constructor(message: string) {
    super(message)
    this.name = 'PaystackError'
  }
}

export type PaystackTransaction = {
  reference: string
  status: string
  currency: string
  amountKobo: number
  requestedAmountKobo: number
  feesKobo: number
  channel: string | null
  paidAt: string | null
}

export interface PaystackClient {
  initialize(input: {
    email: string
    amountKobo: number
    reference: string
    callbackUrl: string
    metadata: Record<string, unknown>
  }): Promise<{ authorizationUrl: string }>
  verify(reference: string): Promise<PaystackTransaction>
}

export function createPaystackClient(secretKey: string, fetchImpl: typeof fetch = fetch): PaystackClient {
  async function call(method: string, path: string, body?: unknown): Promise<Record<string, unknown>> {
    let response: Response
    try {
      response = await fetchImpl(`${BASE_URL}${path}`, {
        method,
        headers: { Authorization: `Bearer ${secretKey}`, 'Content-Type': 'application/json' },
        body: body === undefined ? undefined : JSON.stringify(body),
        signal: AbortSignal.timeout(TIMEOUT_MS),
      })
    } catch (error) {
      throw new PaystackError(`Paystack could not be reached (${error instanceof Error ? error.message : String(error)}).`)
    }
    const json = (await response.json().catch(() => null)) as { status?: boolean; message?: string; data?: unknown } | null
    if (!response.ok || !json?.status) throw new PaystackError(`Paystack refused ${path}: ${json?.message ?? response.status}.`)
    return (json.data ?? {}) as Record<string, unknown>
  }

  return {
    async initialize({ email, amountKobo, reference, callbackUrl, metadata }) {
      const data = await call('POST', '/transaction/initialize', { email, amount: amountKobo, reference, callback_url: callbackUrl, metadata })
      if (typeof data.authorization_url !== 'string') throw new PaystackError('Paystack gave no payment link.')
      return { authorizationUrl: data.authorization_url }
    },

    async verify(reference) {
      const data = await call('GET', `/transaction/verify/${encodeURIComponent(reference)}`)
      return {
        reference: String(data.reference ?? reference),
        status: String(data.status ?? ''),
        currency: String(data.currency ?? ''),
        amountKobo: Number(data.amount ?? 0),
        // Pending check P1: with "Pass fees to customers" on, this is the amount before Paystack's fee.
        requestedAmountKobo: Number(data.requested_amount ?? data.amount ?? 0),
        feesKobo: Number(data.fees ?? 0),
        channel: typeof data.channel === 'string' ? data.channel : null,
        paidAt: typeof data.paid_at === 'string' ? data.paid_at : null,
      }
    },
  }
}
```

```ts
  phone: string
  line1: string
  line2?: string | null
  postcode: string

  const [first = 'Customer', ...rest] = person.name.trim().split(/\s+/)
  return {
    ...place(person),
    first_name: first,
    last_name: rest.join(' ') || first,
    email: person.email,
    phone: person.phone,
    line1: person.line1,
    ...(person.line2 ? { line2: person.line2 } : {}),
    zip: person.postcode,

    name: item.name,
    description: item.name,

    value: item.valueNGN,
    quantity: item.quantity,
    weight: item.weightGrams / 1000,

})

  const carrierName = String(raw.carrier_name ?? 'Courier')
  return {
    rateId: String(raw.rate_id ?? raw.id ?? ''),
    carrierName,
    carrierSlug: String(raw.carrier_slug ?? carrierName.toLowerCase().replace(/[^a-z0-9]+/g, '-')),

    // Pending checks P5 and P6: Terminal's amounts are assumed to be naira.
    amountKobo: toKobo(Number(raw.amount ?? 0)),
  }
}

export function createTerminalClient(secretKey: string, baseUrl: string, fetchImpl: typeof fetch = fetch): TerminalClient {
  async function call(method: string, path: string, body?: unknown, timeoutMs = TIMEOUT_MS): Promise<unknown> {
    let response: Response
    try {
      response = await fetchImpl(`${baseUrl}${path}`, {
        method,
        headers: { Authorization: `Bearer ${secretKey}`, 'Content-Type': 'application/json' },
        body: body === undefined ? undefined : JSON.stringify(body),
        signal: AbortSignal.timeout(timeoutMs),
      })
    } catch (error) {
      throw new TerminalError(`Terminal could not be reached (${error instanceof Error ? error.message : String(error)}).`)
    }
    const json = (await response.json().catch(() => null)) as { status?: boolean; message?: string; data?: unknown } | null
    if (!response.ok || !json || json.status === false) {
      throw new TerminalError(json?.message ? `Terminal said: ${json.message}` : `Terminal answered ${response.status}.`)
    }
    return json.data
  }

  return {
    async quotes(input) {
      const body = {
        pickup_address: input.bookable ? contact(input.pickup) : place(input.pickup),
        delivery_address: input.bookable ? contact(input.delivery) : place(input.delivery),
        parcel: parcelJson(input.parcel),
        currency: 'NGN',
        persist_data: input.bookable,
      }
      const data = await call('POST', '/rates/shipment/quotes', body, QUOTE_TIMEOUT_MS)
      return (Array.isArray(data) ? data : [])
        .map((raw) => rateFrom(raw as Record<string, unknown>))
        .filter((rate) => rate.rateId && rate.amountKobo > 0)
    },

    async getRate(rateId) {
      return rateFrom((await call('GET', `/rates/${encodeURIComponent(rateId)}`)) as Record<string, unknown>)
    },

    async walletBalance() {
      const data = (await call('GET', '/users/wallet-balance')) as { amount?: unknown } | null
      // Pending check P5: the balance is assumed to be naira.
      return { amountKobo: toKobo(Number(data?.amount ?? 0)) }
    },

    async book(rateId) {
      const data = ((await call('POST', '/shipments/pickup', { rate_id: rateId })) ?? {}) as Record<string, unknown>
      const extras = (data.extras ?? {}) as Record<string, unknown>
      const shipmentId = text(data.shipment_id) ?? text(data.id)
      if (!shipmentId) throw new TerminalError('Terminal booked the parcel but gave no shipment ID.')
      return {
        shipmentId,
        trackingNumber: text(extras.tracking_number),
        trackingUrl: text(extras.tracking_url),
        labelUrl: text(extras.shipping_label_url),
      }
    },
  }
}
```

```ts
import { createHmac, timingSafeEqual } from 'node:crypto'

/** How long a delivery price can be used at checkout. */
export const STAMP_LIFE_MS = 30 * 60_000
export const MIN_QUOTE_STAMP_SECRET_LENGTH = 32
const CLOCK_SLACK_MS = 60_000

export type StampedQuote = {
  method: 'courier' | 'flat'
  carrierName: string
  carrierSlug: string
  estimate: string
  priceKobo: number
  state: string
  weightGrams: number
  area: string | null
  issuedAt: number
}

export function quoteStampSecret(): string {
  const secret = process.env.QUOTE_STAMP_SECRET ?? ''
  if (secret.length < MIN_QUOTE_STAMP_SECRET_LENGTH) {
    throw new Error(`QUOTE_STAMP_SECRET must be at least ${MIN_QUOTE_STAMP_SECRET_LENGTH} characters.`)
  }
  return secret
}

const signatureOf = (body: string, secret: string) => createHmac('sha256', secret).update(body).digest('base64url')

/** The quote, readable, followed by an HMAC of it. The browser can see the price but cannot change it. */
export function signQuote(quote: StampedQuote, secret: string): string {
  const body = Buffer.from(JSON.stringify(quote)).toString('base64url')
  return `${body}.${signatureOf(body, secret)}`
}

/** The quote inside a stamp, or null when the stamp was changed, made with another secret, or has lapsed. */
export function readQuote(stamp: unknown, secret: string, now = Date.now()): StampedQuote | null {
  if (typeof stamp !== 'string') return null
  const parts = stamp.split('.')
  if (parts.length !== 2 || !parts[0] || !parts[1]) return null
  const [body, signature] = parts
  const expected = Buffer.from(signatureOf(body, secret))
  const given = Buffer.from(signature)
  if (expected.length !== given.length || !timingSafeEqual(expected, given)) return null
  let quote: StampedQuote
  try {
    quote = JSON.parse(Buffer.from(body, 'base64url').toString('utf8')) as StampedQuote
  } catch {
    return null
  }
  if (typeof quote.issuedAt !== 'number') return null
  if (now - quote.issuedAt > STAMP_LIFE_MS || quote.issuedAt - now > CLOCK_SLACK_MS) return null
  return quote
}
```

```ts
import type { PayloadRequest } from 'payload'

import { Refusal } from '@/refusal'
import { parcelWeightGrams, parseCart, type PricedLine, resolveCart } from '@/shop/cart'
import { shopClients } from '@/shop/clients'
import { flatPriceFor } from '@/shop/flatPrice'
import { toNaira } from '@/shop/money'
import { readShopSettings, type ShopSettingsData } from '@/shop/settings'
import { quoteStampSecret, signQuote, type StampedQuote } from '@/shop/stamp'
import { isNigerianState } from '@/shop/states'
import type { TerminalParcel, TerminalPlace, TerminalRate } from '@/shop/terminal'

export type QuoteOption = { carrierName: string; carrierSlug: string; estimate: string; priceNGN: number; stamp: string }
export type QuotesAnswer = { method: 'courier' | 'flat'; quotes: QuoteOption[] }

type ParcelLine = Pick<PricedLine, 'title' | 'optionName' | 'unitPriceKobo' | 'quantity' | 'weightGrams'>

/** The parcel Terminal is asked about: every line, in the shop's box. */
export function parcelFor(lines: ParcelLine[], packagingId: string): TerminalParcel {
  return {
    packagingId,
    items: lines.map((line) => ({
      name: line.optionName ? `${line.title} (${line.optionName})` : line.title,
      valueNGN: toNaira(line.unitPriceKobo),
      quantity: line.quantity,
      weightGrams: line.weightGrams,
    })),
  }
}

function parsePlace(value: unknown): TerminalPlace {
  const where = (value && typeof value === 'object' ? value : {}) as Record<string, unknown>
  const city = typeof where.city === 'string' ? where.city.trim() : ''
  if (!city || city.length > 100) throw new Refusal('Enter the town or city for delivery.')
  if (!isNigerianState(where.state)) throw new Refusal('Pick the state for delivery.')
  return { city, state: where.state }
}

/** Live courier prices, or none when Terminal fails, is slow or has nothing. Checkout then uses the flat price. */
async function liveRates(req: PayloadRequest, settings: ShopSettingsData, delivery: TerminalPlace, lines: PricedLine[]): Promise<TerminalRate[]> {
  try {
    return await shopClients().terminal.quotes({
      bookable: false,
      pickup: { city: settings.pickup.city, state: settings.pickup.state },
      delivery,
      parcel: parcelFor(lines, settings.packagingId),
    })
  } catch (error) {
    req.payload.logger.warn({ err: error, msg: 'Live delivery prices failed, so the flat price was offered.' })
    return []
  }
}

/** The delivery choices for a cart going to a city and state, each with its stamp. */
export async function deliveryQuotes(req: PayloadRequest, body: Record<string, unknown>): Promise<QuotesAnswer> {
  const lines = await resolveCart(req, parseCart(body.items))
  const delivery = parsePlace(body.address)
  const settings = await readShopSettings(req)
  const weightGrams = parcelWeightGrams(lines, settings.boxWeightGrams)
  const secret = quoteStampSecret()
  const issuedAt = Date.now()
  const stamped = (quote: Omit<StampedQuote, 'state' | 'weightGrams' | 'issuedAt'>): QuoteOption => ({
    carrierName: quote.carrierName,
    carrierSlug: quote.carrierSlug,
    estimate: quote.estimate,
    priceNGN: toNaira(quote.priceKobo),
    stamp: signQuote({ ...quote, state: delivery.state, weightGrams, issuedAt }, secret),
  })

  const rates = await liveRates(req, settings, delivery, lines)
  if (rates.length > 0) {
    return {
      method: 'courier',
      quotes: [...rates]
        .sort((a, b) => a.amountKobo - b.amountKobo)
        .map((rate) =>
          stamped({ method: 'courier', carrierName: rate.carrierName, carrierSlug: rate.carrierSlug, estimate: rate.estimate, priceKobo: rate.amountKobo, area: null }),
        ),
    }
  }
  const flat = flatPriceFor(settings.flatAreas, settings.elsewherePriceNGN, delivery.state)
  if (!flat) throw new Refusal('Delivery prices are not available right now. Please try again later.', 503)
  return {
    method: 'flat',
    quotes: [stamped({ method: 'flat', carrierName: 'Standard delivery', carrierSlug: 'flat', estimate: '', priceKobo: flat.priceKobo, area: flat.area })],
  }
}
```

```ts
import type { PayloadRequest } from 'payload'

import { adminEntryUrl } from '@/hooks/formAlert'
import type { Order } from '@/payload-types'
import { formatNaira, toKobo } from '@/shop/money'
import { readShopSettings } from '@/shop/settings'
import { stateLabel } from '@/shop/states'

export const SHOP_EMAIL_KINDS = [
  'receipt',
  'tracking',
  'staff-paid',
  'staff-mismatch',
  'staff-returned',
  'staff-pickup-failed',
  'staff-charge-failed',
  'staff-weight-claim',
] as const

export type ShopEmailKind = (typeof SHOP_EMAIL_KINDS)[number]
export type ShopEmailInput = { kind: ShopEmailKind; orderId: number; details?: string | null }
type Contact = { email: string | null; phone: string | null }
type BuiltEmail = { audience: 'customer' | 'staff'; subject: string; text: string; html: string }

const escapeHtml = (value: string) =>
  value.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;')

const naira = (value: number | null | undefined) => formatNaira(toKobo(value ?? 0))
const paragraphs = (lines: string[]) => lines.map((line) => `<p>${escapeHtml(line)}</p>`).join('')

function itemLines(order: Order): string[] {
  return (order.items ?? []).map(
    (item) => `${item.quantity} x ${item.title}${item.optionName ? ` (${item.optionName})` : ''}: ${naira((item.unitPriceNGN ?? 0) * (item.quantity ?? 0))}`,
  )
}

function contactLine(contact: Contact, number: string): string | null {
  const ways = [contact.email ? `email ${contact.email}` : null, contact.phone ? `call ${contact.phone}` : null].filter(Boolean)
  if (ways.length === 0) return null
  const joined = ways.join(' or ')
  return `Questions about this order? ${joined.charAt(0).toUpperCase()}${joined.slice(1)}, quoting ${number}.`
}

/** The email for one kind of event on one order. Pure, so the wording is tested without sending anything. */
export function buildShopEmail(order: Order, input: ShopEmailInput, contact: Contact): BuiltEmail {
  const number = order.orderNumber ?? `order ${order.id}`
  const link = adminEntryUrl('orders', order.id)

  const customer = (subject: string, lines: string[]): BuiltEmail => {
    const all = [...lines, ...[contactLine(contact, number)].filter((line): line is string => Boolean(line))]
    return { audience: 'customer', subject, text: `${all.join('\n\n')}\n`, html: paragraphs(all) }
  }
  const staff = (subject: string, lines: string[]): BuiltEmail => {
    const all = [...lines, ...(input.details ? [input.details] : [])]
    return {
      audience: 'staff',
      subject,
      text: `${all.join('\n\n')}\n\nOpen it in the admin: ${link}\n`,
      html: `${paragraphs(all)}<p><a href="${link}">Open it in the admin</a></p>`,
    }
  }

  switch (input.kind) {
    case 'receipt': {
      const first = order.customerName?.trim().split(/\s+/)[0]
      const delivery = order.delivery?.method === 'courier' && order.delivery.carrierName
        ? `Delivery (${order.delivery.carrierName}${order.delivery.estimate ? `, ${order.delivery.estimate}` : ''})`
        : 'Delivery'
      const where = [order.address?.line1, order.address?.city, order.address?.state ? stateLabel(order.address.state) : null].filter(Boolean)
      return customer(`Your Noisy Streetss order ${number}`, [
        `${first ? `Thank you, ${first}.` : 'Thank you.'} Your payment has been received and order ${number} is being prepared.`,
        ...itemLines(order),
        `${delivery}: ${naira(order.deliveryNGN)}`,
        ...(order.paystackFeeNGN ? [`Paystack fee: ${naira(order.paystackFeeNGN)}`] : []),
        `Total paid: ${naira(order.chargedNGN ?? order.totalNGN)}`,
        `Delivering to: ${where.join(', ')}`,
        'A second email will follow with a tracking link once the parcel is booked with the courier.',
      ])
    }
    case 'tracking': {
      const carrier = order.shipment?.carrierName ?? 'the courier'
      return customer(`Your order ${number} has been booked with ${carrier}`, [
        `Order ${number} has been booked with ${carrier} and will be collected soon.`,
        ...(order.shipment?.trackingNumber ? [`Tracking number: ${order.shipment.trackingNumber}`] : []),
        ...(order.shipment?.trackingUrl ? [`Follow it here: ${order.shipment.trackingUrl}`] : []),
      ])
    }
    case 'staff-paid':
      return staff(`New paid order ${number}${order.payment?.paidLate ? ' (paid after it expired)' : ''}`, [
        `${order.customerName ?? 'A customer'} paid ${naira(order.chargedNGN ?? order.totalNGN)} for order ${number}.`,
        ...(order.payment?.paidLate ? ['The payment came in after the order had expired. Check the items are still in stock before packing.'] : []),
        ...itemLines(order),
      ])
    case 'staff-mismatch':
      return staff(`Payment amount problem on ${number}`, [
        `Paystack reported a payment for ${number} that does not match the order, so it was not marked Paid. Check it in Paystack's dashboard.`,
      ])
    case 'staff-returned':
      return staff(`Parcel returned: ${number}`, [`Terminal reports that the parcel for ${number} was returned. Contact the customer, then refund or book again.`])
    case 'staff-pickup-failed':
      return staff(`Pickup failed: ${number}`, [`Terminal reports that the courier could not collect the parcel for ${number}. Check Terminal's dashboard.`])
    case 'staff-charge-failed':
      return staff(`Terminal could not charge the wallet: ${number}`, [
        `Terminal could not take payment from the wallet for ${number}. Top up the wallet in Terminal's dashboard, then check the booking.`,
      ])
    case 'staff-weight-claim':
      return staff(`Heavier parcel claimed: ${number}`, [
        `The carrier recorded a heavier parcel than was booked for ${number}. The extra is billed to the Terminal wallet 7 days after the claim unless it is disputed in Terminal's dashboard.`,
      ])
  }
}

/** Builds and sends one shop email. Run by the sendShopEmail job, so a failure is retried. */
export async function sendShopEmail(req: PayloadRequest, input: ShopEmailInput): Promise<void> {
  const order = await req.payload.findByID({ collection: 'orders', id: input.orderId, depth: 0, overrideAccess: true, disableErrors: true, req })
  if (!order) {
    req.payload.logger.warn(`The "${input.kind}" email was skipped: order ${input.orderId} no longer exists.`)
    return
  }
  const site = await req.payload.findGlobal({ slug: 'site-settings', depth: 0, overrideAccess: true, req })
  const email = buildShopEmail(order, input, { email: site.contactEmail ?? null, phone: site.phone ?? null })
  const to = email.audience === 'customer' ? order.customerEmail : (await readShopSettings(req)).orderAlertEmail
  if (!to) {
    req.payload.logger.warn(`No address for the "${input.kind}" email about ${order.orderNumber}, so it was not sent.`)
    return
  }
  await req.payload.sendEmail({ to, subject: email.subject, text: email.text, html: email.html })
}

export async function queueShopEmail(req: PayloadRequest, input: ShopEmailInput): Promise<void> {
  await req.payload.jobs.queue({ task: 'sendShopEmail', input, req })
}
```

```ts
import { sql } from '@payloadcms/db-postgres'
import type { PayloadRequest } from 'payload'

import type { OrderStatus } from '@/shop/statuses'

type Database = { execute: (query: ReturnType<typeof sql>) => Promise<{ rows: unknown[] }> }

const database = (req: PayloadRequest) => (req.payload.db as unknown as { drizzle: Database }).drizzle

/** A booking that never finished (a crash or a timeout) stops holding its order after this long. */
export const BOOKING_CLAIM_LAPSES_MS = 5 * 60_000

/**
 * Moves an order from one of `from` to `to` in a single statement, and says whether this call did it.
 * Payload's own update with a `where` finds the records first and then changes them, which two
 * callers arriving together could both pass. The statuses come from this code, never from a request.
 */
export async function claimStatus(req: PayloadRequest, id: number, from: readonly OrderStatus[], to: OrderStatus): Promise<boolean> {
  const allowed = sql.join(
    from.map((status) => sql`${status}::enum_orders_status`),
    sql`, `,
  )
  const result = await database(req).execute(sql`
    update "orders" set "status" = ${to}::enum_orders_status, "updated_at" = now()
    where "id" = ${id} and "status" in (${allowed})
    returning "id"`)
  return result.rows.length > 0
}

/** Holds a Paid order for one booking at a time. A hold older than five minutes has lapsed. */
export async function claimBooking(req: PayloadRequest, id: number): Promise<boolean> {
  const lapsed = new Date(Date.now() - BOOKING_CLAIM_LAPSES_MS).toISOString()
  const result = await database(req).execute(sql`
    update "orders" set "shipment_booking_started_at" = now()
    where "id" = ${id} and "status" = 'paid'
      and ("shipment_booking_started_at" is null or "shipment_booking_started_at" < ${lapsed}::timestamptz)
    returning "id"`)
  return result.rows.length > 0
}

export async function releaseBooking(req: PayloadRequest, id: number): Promise<void> {
  await database(req).execute(sql`update "orders" set "shipment_booking_started_at" = null where "id" = ${id}`)
}
```

```ts
import type { PayloadRequest } from 'payload'

import type { Order } from '@/payload-types'
import { claimStatus } from '@/shop/claims'
import { shopClients } from '@/shop/clients'
import { queueShopEmail } from '@/shop/emails'
import { appendHistory } from '@/shop/history'
import { formatNaira, toKobo, toNaira } from '@/shop/money'
import { findOrderByReference, serverUpdate } from '@/shop/orders'
import type { PaystackTransaction } from '@/shop/paystack'

export type MarkPaidResult = 'paid' | 'paid-late' | 'unchanged' | 'not-paid' | 'mismatch' | 'unknown-order'

/** Records an amount that does not match on the order, and alerts staff the first time only. */
async function recordMismatch(req: PayloadRequest, order: Order, payment: PaystackTransaction, expectedKobo: number): Promise<void> {
  if (order.payment?.problem) return
  const problem = `Paystack reported ${formatNaira(payment.requestedAmountKobo)} in ${payment.currency || 'an unknown currency'}, but the order total is ${formatNaira(expectedKobo)}.`
  await serverUpdate(req, order.id, { payment: { ...order.payment, problem } })
  await queueShopEmail(req, { kind: 'staff-mismatch', orderId: order.id, details: problem })
}

/**
 * Marks an order Paid once Paystack itself confirms the payment. Used by Paystack's webhook, the
 * return page and the expiry job, so it is safe to run twice or at the same moment: only the call
 * that actually moves the order records the payment and queues the emails. An Expired order is
 * marked Paid late (decided 27 October 2026).
 */
export async function markPaid(req: PayloadRequest, reference: string, by: string): Promise<MarkPaidResult> {
  const order = await findOrderByReference(req, reference)
  if (!order) return 'unknown-order'
  if (order.status !== 'awaiting-payment' && order.status !== 'expired') return 'unchanged'

  const payment = await shopClients().paystack.verify(reference)
  if (payment.status !== 'success') return 'not-paid'
  const expectedKobo = toKobo(order.totalNGN ?? 0)
  // Pending check P1: requestedAmountKobo is the amount before Paystack added its fee.
  if (payment.currency !== 'NGN' || payment.requestedAmountKobo !== expectedKobo) {
    await recordMismatch(req, order, payment, expectedKobo)
    return 'mismatch'
  }

  const from = order.status
  if (!(await claimStatus(req, order.id, ['awaiting-payment', 'expired'], 'paid'))) return 'unchanged'
  const late = from === 'expired'
  await serverUpdate(req, order.id, {
    payment: { ...order.payment, channel: payment.channel, paidAt: payment.paidAt ?? new Date().toISOString(), paidLate: late },
    chargedNGN: toNaira(payment.amountKobo),
    paystackFeeNGN: toNaira(payment.feesKobo),
    history: appendHistory(order.history, { from, to: 'paid', by, note: late ? 'Paid after the order expired' : null }),
  })
  await queueShopEmail(req, { kind: 'receipt', orderId: order.id })
  await queueShopEmail(req, { kind: 'staff-paid', orderId: order.id })
  return late ? 'paid-late' : 'paid'
}
```

```ts
/** A Nigerian number written as 0 and ten digits, or +234 and ten digits, stored as +234 and ten digits. */
export function normaliseNigerianPhone(value: unknown): string | null {
  if (typeof value !== 'string') return null
  const compact = value.replace(/[\s()-]/g, '')
  if (/^0\d{10}$/.test(compact)) return `+234${compact.slice(1)}`
  if (/^\+234\d{10}$/.test(compact)) return compact
  return null
}
```

```ts
import type { PayloadRequest } from 'payload'

import { claimStatus } from '@/shop/claims'
import { appendHistory } from '@/shop/history'
import { markPaid } from '@/shop/markPaid'
import { findOrder, serverUpdate } from '@/shop/orders'

/** How many times the expiry job tries again when Paystack can't be reached. */
export const EXPIRY_RETRIES = 2

/**
 * Runs when an unpaid order's 30 minutes are up. Paystack is asked once more first, because many
 * bank-transfer payers never come back to the site. If Paystack can't be reached, the job tries
 * again; after the last try the order expires anyway, and a payment found later still marks it Paid.
 */
export async function expireOrder(req: PayloadRequest, orderId: number, triesSoFar: number): Promise<'paid' | 'expired' | 'unchanged'> {
  const order = await findOrder(req, orderId)
  if (!order || order.status !== 'awaiting-payment') return 'unchanged'
  try {
    const result = await markPaid(req, order.payment?.reference ?? '', 'Expiry check')
    if (result === 'paid' || result === 'paid-late') return 'paid'
  } catch (error) {
    if (triesSoFar < EXPIRY_RETRIES) throw error
    req.payload.logger.warn({ err: error, msg: `Paystack could not be asked about ${order.orderNumber} before it expired.` })
  }
  if (!(await claimStatus(req, order.id, ['awaiting-payment'], 'expired'))) return 'unchanged'
  await serverUpdate(req, order.id, {
    history: appendHistory(order.history, { from: 'awaiting-payment', to: 'expired', by: 'Expiry check' }),
  })
  return 'expired'
}
```

```ts
import { randomBytes } from 'node:crypto'
import { commitTransaction, initTransaction, killTransaction, type PayloadRequest } from 'payload'

import { isSpam } from '@/hooks/honeypot'
import { Refusal } from '@/refusal'
import { emailValidator } from '@/rules/validators'
import { parcelWeightGrams, parseCart, priceOrder, resolveCart } from '@/shop/cart'
import { shopClients } from '@/shop/clients'
import { SHOP_SERVER } from '@/shop/context'
import { toNaira } from '@/shop/money'
import { PaystackError } from '@/shop/paystack'
import { normaliseNigerianPhone } from '@/shop/phone'
import { readShopSettings } from '@/shop/settings'
import { quoteStampSecret, readQuote } from '@/shop/stamp'
import { isNigerianState } from '@/shop/states'

export type CheckoutAnswer = { paymentUrl: string; reference: string }

export const STALE_DELIVERY = 'Delivery prices have changed. Refresh them and try again.'

const text = (value: unknown) => (typeof value === 'string' ? value.trim() : '')

/** Where Paystack sends the customer back to. Paystack adds `?trxref=<reference>&reference=<reference>`. */
function returnUrl(): string {
  return `${(process.env.NEXT_PUBLIC_SERVER_URL || 'http://localhost:3000').replace(/\/$/, '')}/checkout/complete`
}

/** Random and unguessable: it sits in the return page's web address and unlocks the order summary. */
const newReference = () => `ns_${randomBytes(12).toString('hex')}`

async function parseCustomer(req: PayloadRequest, body: Record<string, unknown>) {
  const name = text(body.name)
  if (!name || name.length > 120) throw new Refusal('Enter your name.')
  const email = text(body.email).toLowerCase()
  const emailCheck = await emailValidator(email, { req, required: true } as never)
  if (!email || emailCheck !== true) throw new Refusal('Enter a valid email address.')
  const phone = normaliseNigerianPhone(body.phone)
  if (!phone) throw new Refusal('Enter a Nigerian phone number, such as 0803 123 4567.')
  return { name, email, phone }
}

function parseAddress(value: unknown) {
  const address = (value && typeof value === 'object' ? value : {}) as Record<string, unknown>
  const line1 = text(address.line1)
  const line2 = text(address.line2)
  const city = text(address.city)
  const postcode = text(address.postcode)
  if (!line1 || line1.length > 200 || line2.length > 200) throw new Refusal('Enter the street address for delivery.')
  if (!city || city.length > 100) throw new Refusal('Enter the town or city for delivery.')
  if (!isNigerianState(address.state)) throw new Refusal('Pick the state for delivery.')
  // Pending check P2: optional until the sandbox shows whether Terminal needs it at checkout.
  if (postcode && !/^\d{6}$/.test(postcode)) throw new Refusal('A Nigerian postcode has six digits. Leave it empty if you are not sure.')
  return { line1, line2: line2 || null, city, state: address.state, postcode: postcode || null }
}

/**
 * Creates the order and starts its Paystack payment, all or nothing: if Paystack can't start the
 * payment, no order is kept. Everything the customer sees is checked and priced again here.
 */
export async function checkout(req: PayloadRequest, body: Record<string, unknown>): Promise<CheckoutAnswer> {
  if (isSpam(body)) {
    // As B1's forms do: a bot gets a normal-looking answer, nothing is saved and Paystack is never called.
    const reference = newReference()
    return { paymentUrl: `${returnUrl()}?reference=${reference}`, reference }
  }
  const customer = await parseCustomer(req, body)
  const address = parseAddress(body.address)
  const lines = await resolveCart(req, parseCart(body.items))
  const settings = await readShopSettings(req)
  const quote = readQuote((body.delivery as { stamp?: unknown } | undefined)?.stamp, quoteStampSecret())
  if (!quote || quote.state !== address.state || quote.weightGrams !== parcelWeightGrams(lines, settings.boxWeightGrams)) {
    throw new Refusal(STALE_DELIVERY)
  }
  const totals = priceOrder(lines, quote.priceKobo)
  const reference = newReference()
  const now = new Date()
  const expiresAt = new Date(now.getTime() + settings.expiryMinutes * 60_000)

  const shouldCommit = await initTransaction(req)
  try {
    const order = await req.payload.create({
      collection: 'orders',
      overrideAccess: true,
      context: { [SHOP_SERVER]: true },
      depth: 0,
      req,
      data: {
        status: 'awaiting-payment',
        customerName: customer.name,
        customerEmail: customer.email,
        customerPhone: customer.phone,
        address: { ...address, country: 'Nigeria' },
        items: lines.map((line) => ({
          product: { relationTo: line.collection, value: line.id },
          optionId: line.optionId,
          optionName: line.optionName,
          title: line.title,
          unitPriceNGN: toNaira(line.unitPriceKobo),
          quantity: line.quantity,
          weightGrams: line.weightGrams,
        })),
        delivery: {
          method: quote.method,
          carrierName: quote.carrierName,
          carrierSlug: quote.carrierSlug,
          estimate: quote.estimate || null,
          priceNGN: toNaira(quote.priceKobo),
          flatArea: quote.area,
        },
        itemsNGN: toNaira(totals.itemsKobo),
        deliveryNGN: toNaira(totals.deliveryKobo),
        totalNGN: toNaira(totals.totalKobo),
        payment: { reference },
        expiresAt: expiresAt.toISOString(),
        history: [{ at: now.toISOString(), to: 'awaiting-payment', by: 'Checkout' }],
      },
    })
    const { authorizationUrl } = await shopClients().paystack.initialize({
      email: customer.email,
      amountKobo: totals.totalKobo,
      reference,
      callbackUrl: returnUrl(),
      metadata: { orderId: order.id, orderNumber: order.orderNumber },
    })
    if (shouldCommit) await commitTransaction(req)
    await req.payload.jobs.queue({ task: 'expireOrder', input: { orderId: order.id }, waitUntil: expiresAt, req })
    return { paymentUrl: authorizationUrl, reference }
  } catch (error) {
    await killTransaction(req)
    if (error instanceof PaystackError) {
      req.payload.logger.warn({ err: error, msg: 'Paystack could not start a payment, so no order was kept.' })
      throw new Refusal('The payment could not be started. Please try again in a moment.', 502)
    }
    throw error
  }
}
```

```ts
/** How long an Expired order keeps the customer's details. */
export const DETAILS_KEPT_DAYS = 30

/**
 * Empties the name, email, phone and street lines on orders Expired for more than 30 days, keeping
 * the city, state, items and totals. Nigeria's Data Protection Act 2023 expects personal data to be
 * kept no longer than its purpose needs, and an unpaid order has none left. Returns how many it cleaned.
 */
export async function removeExpiredDetails(req: PayloadRequest, now = new Date()): Promise<number> {
  const cutoff = new Date(now.getTime() - DETAILS_KEPT_DAYS * 24 * 60 * 60 * 1000).toISOString()
  const { docs } = await req.payload.find({
    collection: 'orders',
    where: { and: [{ status: { equals: 'expired' } }, { expiresAt: { less_than: cutoff } }, { detailsRemovedAt: { exists: false } }] },
    depth: 0,
    limit: 0,
    pagination: false,
    overrideAccess: true,
    req,
  })
  for (const order of docs) {
    await serverUpdate(req, order.id, {
      customerName: null,
      customerEmail: null,
      customerPhone: null,
      address: { ...order.address, line1: null, line2: null, postcode: null },
      detailsRemovedAt: now.toISOString(),
      history: appendHistory(order.history, {
        by: 'Data clean-up',
        note: `Customer details removed ${DETAILS_KEPT_DAYS} days after the order expired`,
      }),
    })
  }
  return docs.length
}
```

```ts
import { commitTransaction, initTransaction, killTransaction, type PayloadRequest } from 'payload'

import type { Book, Merch } from '@/payload-types'
import { Refusal } from '@/refusal'

export type StockRow = {
  key: string
  collection: 'books' | 'merch'
  id: number
  title: string
  optionId: string | null
  optionName: string | null
  inStock: boolean
}

export type StockChange = { collection: 'books' | 'merch'; id: number; optionId: string | null; inStock: boolean }

/** Every format and size on sale on the live site, with its In stock value. */
export async function stockRows(req: PayloadRequest): Promise<StockRow[]> {
  const live = { draft: false, depth: 0, limit: 0, pagination: false, sort: 'title', overrideAccess: true, req } as const
  const books = await req.payload.find({ collection: 'books', where: { and: [{ sellInShop: { equals: true } }, { _status: { equals: 'published' } }] }, ...live })
  const merch = await req.payload.find({ collection: 'merch', where: { _status: { equals: 'published' } }, ...live })
  const rows: StockRow[] = []
  for (const book of books.docs) {
    for (const format of book.formats ?? []) {
      rows.push({ key: `books:${book.id}:${format.id}`, collection: 'books', id: book.id, title: book.title, optionId: format.id ?? null, optionName: format.formatName, inStock: Boolean(format.inStock) })
    }
  }
  for (const item of merch.docs) {
    const options = item.options ?? []
    if (options.length === 0) {
      rows.push({ key: `merch:${item.id}`, collection: 'merch', id: item.id, title: item.title, optionId: null, optionName: null, inStock: Boolean(item.inStock) })
    }
    for (const option of options) {
      rows.push({ key: `merch:${item.id}:${option.id}`, collection: 'merch', id: item.id, title: item.title, optionId: option.id ?? null, optionName: option.name, inStock: Boolean(option.inStock) })
    }
  }
  return rows
}

/** The fields to change for one In stock value, or null when the option is no longer on the record. */
function withStock(doc: Book | Merch, change: StockChange): Record<string, unknown> | null {
  if (change.collection === 'books') {
    const formats = (doc as Book).formats ?? []
    if (!formats.some((format) => format.id === change.optionId)) return null
    return { formats: formats.map((format) => (format.id === change.optionId ? { ...format, inStock: change.inStock } : format)) }
  }
  const options = (doc as Merch).options ?? []
  if (options.length === 0) return change.optionId ? null : { inStock: change.inStock }
  if (!options.some((option) => option.id === change.optionId)) return null
  return { options: options.map((option) => (option.id === change.optionId ? { ...option, inStock: change.inStock } : option)) }
}

/** The whole record as save data, without what Payload sets itself. */
function asData(doc: Book | Merch): Record<string, unknown> {
  const { id: _id, createdAt: _createdAt, updatedAt: _updatedAt, ...rest } = doc as unknown as Record<string, unknown>
  return rest
}

function find(req: PayloadRequest, change: StockChange, draft: boolean): Promise<Book | Merch | null> {
  const options = { id: change.id, draft, depth: 0, overrideAccess: true, disableErrors: true, req } as const
  return change.collection === 'books' ? req.payload.findByID({ collection: 'books', ...options }) : req.payload.findByID({ collection: 'merch', ...options })
}

/**
 * Changes one In stock value on the live site. In Payload 3.90.2, publishing by id builds on the
 * newest version, draft included, so the whole published record is sent (nothing from a draft is
 * merged in). If an editor's draft was newer, it is then saved again as a draft with the same value,
 * so it stays the newest version and can't undo the switch when it's published.
 */
export async function setStock(req: PayloadRequest, change: StockChange): Promise<void> {
  const published = await find(req, change, false)
  if (!published || published._status !== 'published') throw new Refusal('Only items on the live site can have their stock changed here.')
  const liveChange = withStock(published, change)
  if (!liveChange) throw new Refusal('That option no longer exists. Reload the page.')
  const latest = await find(req, change, true)
  const draftWaiting = latest?._status === 'draft'

  const shouldCommit = await initTransaction(req)
  try {
    const save = { collection: change.collection, id: change.id, overrideAccess: true, depth: 0, req }
    await req.payload.update({ ...save, draft: false, data: { ...asData(published), ...liveChange, _status: 'published' } as never })
    if (draftWaiting && latest) {
      await req.payload.update({ ...save, draft: true, data: { ...asData(latest), ...(withStock(latest, change) ?? {}), _status: 'draft' } as never })
    }
    if (shouldCommit) await commitTransaction(req)
  } catch (error) {
    await killTransaction(req)
    throw error
  }
}

/** A switch from the Stock page, checked for shape. */
export function parseStockChange(body: Record<string, unknown>): StockChange {
  const id = Number(body.id)
  const optionId = body.optionId === undefined || body.optionId === null || body.optionId === '' ? null : body.optionId
  if ((body.collection !== 'books' && body.collection !== 'merch') || !Number.isInteger(id) || id < 1 || typeof body.inStock !== 'boolean' || (optionId !== null && typeof optionId !== 'string')) {
    throw new Refusal('The change could not be read. Reload the page.')
  }
  return { collection: body.collection, id, optionId, inStock: body.inStock }
}
```

```tsx
'use client'

import { useMemo, useState } from 'react'

import type { StockRow } from '@/shop/stock'

type Group = { key: string; title: string; rows: StockRow[] }

const card = {
  border: '1px solid var(--theme-elevation-150)',
  borderRadius: 4,
  padding: '12px 16px',
  marginBottom: 12,
  background: 'var(--theme-elevation-0)',
}

const chip = (inStock: boolean) => ({
  display: 'inline-flex',
  alignItems: 'center',
  gap: 8,
  padding: '6px 10px',
  border: `1px solid ${inStock ? 'var(--theme-elevation-200)' : 'var(--theme-error-400)'}`,
  borderRadius: 4,
  cursor: 'pointer',
})

/** The Stock page's search box and switches. Each switch saves straight away. */
export function StockList({ rows: initial }: { rows: StockRow[] }) {
  const [rows, setRows] = useState(initial)
  const [query, setQuery] = useState('')
  const [busy, setBusy] = useState<string | null>(null)
  const [error, setError] = useState<string | null>(null)

  const groups = useMemo(() => {
    const wanted = query.trim().toLowerCase()
    const byItem = new Map<string, Group>()
    for (const row of rows) {
      if (wanted && !row.title.toLowerCase().includes(wanted)) continue
      const key = `${row.collection}:${row.id}`
      const group = byItem.get(key) ?? { key, title: row.title, rows: [] }
      group.rows.push(row)
      byItem.set(key, group)
    }
    return [...byItem.values()]
  }, [rows, query])

  async function toggle(row: StockRow) {
    setBusy(row.key)
    setError(null)
    const inStock = !row.inStock
    const res = await fetch('/api/shop/stock', {
      method: 'POST',
      credentials: 'include',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ collection: row.collection, id: row.id, optionId: row.optionId, inStock }),
    })
    if (res.ok) {
      setRows((current) => current.map((candidate) => (candidate.key === row.key ? { ...candidate, inStock } : candidate)))
    } else {
      const json = (await res.json().catch(() => null)) as { errors?: { message?: string }[] } | null
      setError(json?.errors?.[0]?.message ?? 'The change was not saved. Try again.')
    }
    setBusy(null)
  }

  return (
    <div>
      <input
        aria-label="Search by title"
        placeholder="Search by title"
        value={query}
        onChange={(event) => setQuery(event.target.value)}
        style={{ width: '100%', maxWidth: 480, padding: '8px 12px', marginBottom: 16, border: '1px solid var(--theme-elevation-150)', borderRadius: 4 }}
      />
      {error ? (
        <p role="alert" style={{ color: 'var(--theme-error-500)', marginBottom: 12 }}>
          {error}
        </p>
      ) : null}
      {groups.length === 0 ? <p>Nothing on sale matches that title.</p> : null}
      {groups.map((group) => (
        <div key={group.key} style={card}>
          <div style={{ fontWeight: 600, marginBottom: 8 }}>{group.title}</div>
          <div style={{ display: 'flex', flexWrap: 'wrap', gap: 8 }}>
            {group.rows.map((row) => (
              <label key={row.key} style={chip(row.inStock)}>
                <input type="checkbox" checked={row.inStock} disabled={busy === row.key} onChange={() => void toggle(row)} />
                <span>{row.optionName ?? 'In stock'}</span>
                {row.inStock ? null : <span style={{ color: 'var(--theme-error-500)' }}>Sold out</span>}
              </label>
            ))}
          </div>
        </div>
      ))}
    </div>
  )
}
```

```ts
/** A bookable quote: full addresses for both ends, and every line of the order in the shop's box. */
function bookableRequest(order: Order, settings: ShopSettingsData): Extract<QuoteRequest, { bookable: true }> {

    bookable: true,

      line1: settings.pickup.line1,

        optionName: item.optionName ?? null,
        unitPriceKobo: toKobo(item.unitPriceNGN ?? 0),

        weightGrams: item.weightGrams ?? 0,
      })),

    ),
  }

const reasonOf = (error: unknown) => (error instanceof Error ? error.message : 'Unknown error')

/**
 * What the Book courier panel shows. Terminal doesn't say how long a quote stays bookable, so
 * today's bookable quotes are fetched each time the panel opens. Nothing is asked of Terminal for
 * an order that can't be booked yet.
 */
export async function courierOptions(req: PayloadRequest, orderId: number): Promise<CourierOptions> {
  const order = await findOrder(req, orderId)
  if (!order) throw new Refusal('That order no longer exists.', 404)
  const shipment = order.shipment
  const startedAt = shipment?.bookingStartedAt
  const customerChoice =
    order.delivery?.method === 'courier' && order.delivery.carrierSlug ? { carrierName: order.delivery.carrierName ?? '', carrierSlug: order.delivery.carrierSlug } : null
  const base: CourierOptions = {
    status: order.status,
    booked: shipment?.terminalShipmentId
      ? { carrierName: shipment.carrierName ?? null, trackingNumber: shipment.trackingNumber ?? null, trackingUrl: shipment.trackingUrl ?? null, labelUrl: shipment.labelUrl ?? null }
      : null,
    customerChoice,
    paidDeliveryKobo: toKobo(order.deliveryNGN ?? 0),
    walletKobo: null,
    missingPostcode: !order.address?.postcode,
    staleClaim: Boolean(startedAt) && Date.now() - Date.parse(startedAt as string) > BOOKING_CLAIM_LAPSES_MS,
    options: [],
    terminalError: null,
  }
  if (order.status !== 'paid' || base.missingPostcode) return base
  try {
    const { terminal } = shopClients()
    const settings = await readShopSettings(req)
    const [rates, wallet] = await Promise.all([terminal.quotes(bookableRequest(order, settings)), terminal.walletBalance()])
    const options = [...rates]
      .sort((a, b) => a.amountKobo - b.amountKobo)
      .map((rate) => ({ rateId: rate.rateId, carrierName: rate.carrierName, carrierSlug: rate.carrierSlug, estimate: rate.estimate, priceKobo: rate.amountKobo, isCustomerChoice: rate.carrierSlug === customerChoice?.carrierSlug }))
      // The customer's courier first; the rest stay cheapest first.
      .sort((a, b) => Number(b.isCustomerChoice) - Number(a.isCustomerChoice))
    return { ...base, walletKobo: wallet.amountKobo, options }
  } catch (error) {
    req.payload.logger.warn({ err: error, msg: `Courier prices for ${order.orderNumber} could not be fetched.` })
    return { ...base, terminalError: `Couldn't reach Terminal, try again. (${reasonOf(error)})` }
  }
}

/**
 * Books a Paid order with Terminal. The order is claimed first in a single statement, so a double
 * click or a second staff member can't book it twice. On a refusal the claim is released; if the
 * booking went through but recording it failed, the claim is kept and lapses after five minutes,
 * when the panel warns staff to check Terminal's dashboard first.
 */
export async function bookCourier(req: PayloadRequest, orderId: number, rateId: string, bookedBy: string): Promise<Order> {
  const order = await findOrder(req, orderId)
  if (!order) throw new Refusal('That order no longer exists.', 404)
  if (order.status !== 'paid') throw new Refusal('Only a Paid order can be booked.')
  if (!order.address?.postcode) throw new Refusal("Add the customer's postcode to the address first. Couriers need it.")
  if (!(await claimBooking(req, order.id))) throw new Refusal('This order is already being booked. Wait a moment, then reload the page.', 409)

  let bookedWithTerminal = false
  try {
    const { terminal } = shopClients()
    const rate = await terminal.getRate(rateId)
    const wallet = await terminal.walletBalance()
    if (wallet.amountKobo < rate.amountKobo) {
      throw new Refusal(
        `The Terminal wallet holds ${formatNaira(wallet.amountKobo)} and this booking costs ${formatNaira(rate.amountKobo)}. Top it up by at least ${formatNaira(rate.amountKobo - wallet.amountKobo)} in Terminal's dashboard.`,
      )
    }
    const booking = await terminal.book(rate.rateId)
    bookedWithTerminal = true
    const updated = await serverUpdate(req, order.id, {
      status: 'courier-booked',
      shipment: {
        terminalShipmentId: booking.shipmentId,
        carrierName: rate.carrierName,
        trackingNumber: booking.trackingNumber,
        trackingUrl: booking.trackingUrl,
        labelUrl: booking.labelUrl,
        courierCostNGN: toNaira(rate.amountKobo),
        bookedAt: new Date().toISOString(),
        bookedBy,
        bookingStartedAt: null,
      },
      history: appendHistory(order.history, { from: 'paid', to: 'courier-booked', by: bookedBy, note: `Booked with ${rate.carrierName} for ${formatNaira(rate.amountKobo)}` }),
    })
    await queueShopEmail(req, { kind: 'tracking', orderId: order.id })
    return updated
  } catch (error) {
    if (bookedWithTerminal) {
      req.payload.logger.error({ err: error, msg: `${order.orderNumber} was booked with Terminal but the booking could not be recorded. Check Terminal's dashboard.` })
      throw error
    }
    await releaseBooking(req, order.id)
    if (error instanceof TerminalError) throw new Refusal(`Terminal refused the booking: ${error.message}`, 502)
    throw error
  }
}
```

```ts
import type { CollectionConfig, PayloadRequest } from 'payload'

import { nobody } from '@/access'

/** Event ids already handled, so a notice sent twice is acted on once. Server code only. */
export const WebhookEvents: CollectionConfig = {
  slug: 'webhook-events',
  labels: { singular: 'Webhook event', plural: 'Webhook events' },
  admin: { hidden: true },
  access: { read: nobody, create: nobody, update: nobody, delete: nobody },
  fields: [
    { name: 'key', type: 'text', required: true, unique: true, index: true },
    { name: 'receivedAt', type: 'date', required: true },
  ],
}

/** True the first time an event id arrives; false for a repeat. The unique key settles a race. */
export async function firstTimeSeen(req: PayloadRequest, provider: string, eventId: string): Promise<boolean> {
  const key = `${provider}:${eventId}`
  try {
    await req.payload.create({ collection: 'webhook-events', data: { key, receivedAt: new Date().toISOString() }, overrideAccess: true, depth: 0, req })
    return true
  } catch (error) {
    const { totalDocs } = await req.payload.count({ collection: 'webhook-events', where: { key: { equals: key } }, overrideAccess: true, req })
    if (totalDocs > 0) return false
    throw error
  }
}
```

```ts
import type { PayloadRequest } from 'payload'

import type { Order } from '@/payload-types'
import { queueShopEmail, type ShopEmailKind } from '@/shop/emails'
import { appendHistory } from '@/shop/history'
import { formatNaira, toKobo } from '@/shop/money'
import { serverUpdate } from '@/shop/orders'
import { isForwardMove, type OrderStatus, statusLabel } from '@/shop/statuses'

export type TerminalEvent = { type: string; shipmentId: string | null; data: Record<string, unknown> }

const objectOf = (value: unknown) => (value && typeof value === 'object' ? (value as Record<string, unknown>) : {})
const textOf = (value: unknown) => (typeof value === 'string' && value ? value : null)

/** The event's id, name and shipment id, wherever Terminal puts them. */
export function readTerminalEvent(body: unknown): { eventId: string | null; event: TerminalEvent } {
  const raw = objectOf(body)
  const data = objectOf(raw.data)
  const shipment = objectOf(data.shipment)
  return {
    eventId: textOf(raw.event_id) ?? textOf(raw.id),
    event: {
      type: textOf(raw.event) ?? textOf(raw.type) ?? '',
      shipmentId: textOf(data.shipment_id) ?? textOf(shipment.shipment_id) ?? textOf(data.id),
      data,
    },
  }
}

const MOVES: Record<string, OrderStatus> = {
  'pickup.completed': 'in-transit',
  'shipment.in-transit': 'in-transit',
  'shipment.delivered': 'delivered',
  'shipment.returned': 'returned',
}

const ALERTS: Record<string, ShopEmailKind> = {
  'shipment.returned': 'staff-returned',
  'pickup.failed': 'staff-pickup-failed',
  'transaction.failed': 'staff-charge-failed',
  'reconciliation.flagged': 'staff-weight-claim',
}

const CANCELLED = new Set(['shipment.cancelled', 'pickup.cancelled'])

const note = (req: PayloadRequest, order: Order, text: string) =>
  serverUpdate(req, order.id, { history: appendHistory(order.history, { by: 'Terminal', note: text }) })

function alertDetails(event: TerminalEvent): string | null {
  return typeof event.data.amount === 'number' ? `Amount claimed: ${formatNaira(toKobo(event.data.amount))}.` : null
}

/** Applies one Terminal event to its order. Webhooks only move an order forward; a cancellation sends it back to Paid. */
export async function applyTerminalEvent(
  req: PayloadRequest,
  event: TerminalEvent,
): Promise<'moved' | 'noted' | 'alerted' | 'cancelled' | 'ignored' | 'unknown-shipment'> {
  if (!MOVES[event.type] && !ALERTS[event.type] && !CANCELLED.has(event.type)) return 'ignored'
  const { docs } = event.shipmentId
    ? await req.payload.find({ collection: 'orders', where: { 'shipment.terminalShipmentId': { equals: event.shipmentId } }, depth: 0, limit: 1, overrideAccess: true, req })
    : { docs: [] as Order[] }
  const order = docs[0]
  if (!order) {
    req.payload.logger.warn(`Terminal sent ${event.type} for shipment ${event.shipmentId ?? 'with no ID'}, which no order has.`)
    return 'unknown-shipment'
  }

  if (CANCELLED.has(event.type)) {
    if (order.status !== 'courier-booked') {
      await note(req, order, `Terminal reported ${event.type} while the order was ${statusLabel(order.status)}. Nothing was changed.`)
      return 'noted'
    }
    const was = order.shipment
    await serverUpdate(req, order.id, {
      status: 'paid',
      shipment: { terminalShipmentId: null, carrierName: null, trackingNumber: null, trackingUrl: null, labelUrl: null, courierCostNGN: null, bookedAt: null, bookedBy: null, bookingStartedAt: null },
      history: appendHistory(order.history, {
        from: 'courier-booked',
        to: 'paid',
        by: 'Terminal',
        note: `Booking cancelled in Terminal (was ${was?.carrierName ?? 'a courier'}, shipment ${was?.terminalShipmentId}, tracking ${was?.trackingNumber ?? 'none'})`,
      }),
    })
    return 'cancelled'
  }

  let result: 'moved' | 'noted' | 'alerted' = 'alerted'
  const to = MOVES[event.type]
  if (to) {
    if (isForwardMove(order.status, to)) {
      await serverUpdate(req, order.id, { status: to, history: appendHistory(order.history, { from: order.status, to, by: 'Terminal' }) })
      result = 'moved'
    } else {
      await note(req, order, `Late update ignored: Terminal reported ${event.type} while the order was ${statusLabel(order.status)}.`)
      result = 'noted'
    }
  }
  const alert = ALERTS[event.type]
  if (alert) await queueShopEmail(req, { kind: alert, orderId: order.id, details: alertDetails(event) })
  return result
}
```
